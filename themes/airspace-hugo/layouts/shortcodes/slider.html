{{/* Params:

     dir: folder of images relative to /assets or /static (e.g. "photos/trip-2025")

     interval: milliseconds between slides (default 3000)

     ratio: CSS aspect-ratio (default "16/9")

*/}}


{{- $dir := .Get "dir" | default "photos" -}}

{{- $interval := .Get "interval" | default "3000" -}}

{{- $ratio := .Get "ratio" | default "16/9" -}}

{{- $pattern := printf "%s/*.{jpg,jpeg,png,webp,avif,gif,svg}" $dir -}}


{{/* Try page bundle, then /assets via Pipes */}}

{{- $imgs := .Page.Resources.Match $pattern -}}

{{- if not (gt (len $imgs) 0) -}}

  {{- $imgs = resources.Match $pattern -}}

{{- end -}}


{{/* If still none, fall back to /static */}}

{{- $static := slice -}}

{{- if not (gt (len $imgs) 0) -}}

  {{- with (readDir (printf "static/%s" $dir)) -}}

    {{- range . -}}

      {{- $n := lower .Name -}}

      {{- if or (hasSuffix $n ".jpg") (hasSuffix $n ".jpeg") (hasSuffix $n ".png") (hasSuffix $n ".webp") (hasSuffix $n ".avif") (hasSuffix $n ".gif") (hasSuffix $n ".svg") -}}

        {{- $static = $static | append (printf "/%s/%s" $dir .Name) -}}

      {{- end -}}

    {{- end -}}

  {{- end -}}

{{- end -}}


{{- $count := cond (gt (len $imgs) 0) (len $imgs) (len $static) -}}


{{- if eq $count 0 -}}

  <div class="hgx-slider-error">No images found in <code>{{ $dir }}</code>.</div>

{{- else -}}


  {{/* Emit a tiny external CSS file so CSP won't block it */}}

  {{- $css := resources.FromString "shortcodes/hgx-slider.css" `

.hgx-slider{position:relative;width:100%;aspect-ratio:var(--ratio,16/9);overflow:hidden;border-radius:12px}

.hgx-track{display:flex;width:calc(100% * var(--count));height:100%;animation:hgx-step calc(var(--count) * var(--interval)) steps(var(--count)) infinite}

.hgx-slider:hover .hgx-track{animation-play-state:paused}

@media (prefers-reduced-motion:reduce){.hgx-track{animation:none}}

.hgx-item{flex:0 0 100%;height:100%}

.hgx-item img{width:100%;height:100%;object-fit:cover;display:block}

.hgx-track.no-anim{animation:none}

@keyframes hgx-step{to{transform:translateX(calc(-100% * (var(--count) - 1)))}}` | minify | fingerprint -}}

  <link rel="stylesheet" href="{{ $css.RelPermalink }}" integrity="{{ $css.Data.Integrity }}">


  <div class="hgx-slider" style="--count:{{ $count }}; --interval:{{ $interval }}ms; --ratio:{{ $ratio }};">

    <div class="hgx-track{{ if le $count 1 }} no-anim{{ end }}">

      {{- if gt (len $imgs) 0 -}}

        {{- range $r := $imgs -}}

          {{- $mt := $r.MediaType -}}

          {{- $out := $r -}}

          {{- if and (eq $mt.Type "image") (ne $mt.SubType "svg") (ne $mt.SubType "gif") -}}

            {{- $out = $r.Fit "1600x q85" -}}

          {{- end -}}

          <figure class="hgx-item"><img loading="lazy" src="{{ $out.RelPermalink }}" alt="{{ $r.Name }}"></figure>

        {{- end -}}

      {{- else -}}

        {{- range $p := $static -}}

          <figure class="hgx-item"><img loading="lazy" src="{{ $p }}" alt=""></figure>

        {{- end -}}

      {{- end -}}

    </div>

  </div>

{{- end -}}
